/*
  WARNING:
  This computation uses bnfinit without bnfcertify.
  Therefore the output should be regarded as conditional on GRH,
  exactly as stated in Section 6 of the paper.
*/

/*
  Section 6, Case (I): corrected element-partition-method search for

      N = p1 * p2,
      p1 == 3 mod 8,
      p2 == 5 mod 8,
      kronecker(p1,p2) = -1.

  K_cyc = Q(zeta_N).
  D is the decomposition field of 2 in K_cyc.
  L is the unique subfield of K_cyc such that [L:D] = 2.

  For odd n >= 3, the program solves in O_L:

      I * conjugate(I) = 2^n O_L.

  Important correction:

  We do not use allshifts(...) to recover all solutions from a
  representative. Instead, for each representative ideal I, we apply
  every automorphism in Gal(L/Q), then recover the exponent vector
  by idealval at the primes in dec.

  This avoids producing fake solutions that are not principal.

  Important decomposition check in this case:

  Since L/D has degree 2 and D is the decomposition field of 2, the
  primes above 2 in L have residue degree 2. Hence the correct check is

      [L:Q] = 2 * #(prime ideals of O_L above 2),

  not [L:Q] = #(prime ideals above 2).
*/


/* ===== Input ===== */

p1 = 19;
p2 = 13;

start_n = 3;
max_n = 101;

default(realprecision, 100);


/* ===== Basic validation ===== */

if(!isprime(p1), error("p1 must be prime"));
if(!isprime(p2), error("p2 must be prime"));
if(p1 == p2, error("p1 and p2 must be distinct"));

if(p1 % 8 != 3, error("p1 must be congruent to 3 modulo 8"));
if(p2 % 8 != 5, error("p2 must be congruent to 5 modulo 8"));

if(kronecker(p1, p2) != -1, error("kronecker(p1,p2) must be -1"));

if(start_n < 1, error("start_n must be positive"));
if(start_n % 2 == 0, error("start_n must be odd"));
if(max_n < start_n, error("max_n must be at least start_n"));

N = p1 * p2;


/* ===== Compute m_E and the correct norm-pruning divisor ===== */

/*
  In Case (I) of the endpoint-extension theorem, m_E is the smallest
  positive odd integer
  such that

      p2*y^2 + p1*z^2 = 2^(m_E+2)

  has an integral solution.

  The order of a prime ideal P above 2 in E = Q(sqrt(-p1*p2)) is 2*m_E.

  But in L/E, the norm of each prime of L above P is P^2. Hence the
  pruning condition is

      A - B == 0 mod m_E.

  Therefore norm_pruning_divisor = m_E.
*/

has_solution_case1_E(p1, p2, m) =
{
  my(T = 2^(m + 2));
  my(ymax = sqrtint(T \ p2));
  my(y, rem, z2);

  for(y = 0, ymax,
    rem = T - p2*y^2;

    if(rem >= 0 && rem % p1 == 0,
      z2 = rem / p1;

      if(issquare(z2),
        return(1);
      );
    );
  );

  return(0);
}


compute_m_case1_E(p1, p2) =
{
  my(m = 1);

  while(1,
    if(has_solution_case1_E(p1, p2, m),
      return(m);
    );

    m += 2;
  );
}


m_E = compute_m_case1_E(p1, p2);
ord_2_in_E = 2 * m_E;
norm_pruning_divisor = m_E;


/* ===== Construct L/E ===== */

/*
  D is fixed by <2>.
  L is fixed by <4>.

  Since <4> is an index-2 subgroup of <2>, we have [L:D] = 2.
*/

phi_N = eulerphi(N);
ord4_N = znorder(Mod(4, N));

if(phi_N % ord4_N != 0, error("Unexpected: ord_N(4) does not divide phi(N)."));

degL_expected = phi_N / ord4_N;

fL0 = galoissubcyclo(N, 4);

if(poldegree(fL0) != degL_expected, error("Degree of fL0 is not phi(N)/ord_N(4)."));


/*
  E = Q(sqrt(-p1*p2)).
  We use y as the variable of E and x as the variable of L.
*/

Enf = nfinit(y^2 + p1*p2);


/*
  Factor the absolute polynomial fL0 over E.
  One irreducible factor gives a relative defining polynomial for L/E.
*/

fac_LE = nffactor(Enf, fL0);
relpol_LE = fac_LE[1, 1];


/*
  Use flag 1 so that the rnf structure contains an attached absolute
  number field structure for L.
*/

rnf_LE = rnfinit(Enf, relpol_LE, 1);
Labs = nfinit(rnf_LE);
Lbnf = bnfinit(Labs);

dec = idealprimedec(Labs, 2);


/*
  Degree/decomposition sanity checks.

  In this case the expected relation is

      [L:Q] = 2 * #dec,

  because the residue degree of primes above 2 in L is 2.
*/

if(poldegree(Labs.pol) != degL_expected, error("The constructed field L has the wrong degree."));
if(2 * #dec != degL_expected, error("Unexpected decomposition of 2 in L: 2*#dec is not equal to [L:Q]."));


/* ===== Compute all Galois automorphisms of L/Q ===== */

G_all = galoisinit(Labs);
all_galois_automorphisms = Vec(galoispermtopol(G_all, G_all.group));

if(#all_galois_automorphisms != poldegree(Labs.pol), error("The number of automorphisms is not equal to [L:Q]; L/Q does not appear to be Galois as required."));


/* ===== Find complex conjugation on L ===== */

eval_aut_at_complex_root(aut, nf, gamma) =
{
  my(v = variable(nf.pol));
  subst(lift(aut), v, gamma);
}


choose_nonreal_root(nf) =
{
  my(rr = nf.roots);

  for(i = 1, #rr,
    if(abs(imag(rr[i])) > 10^(-30),
      return(rr[i]);
    );
  );

  error("No numerically nonreal root found. Increase realprecision or check the field.");
}


find_complex_conjugation_from_auts(nf, auts) =
{
  my(gamma = choose_nonreal_root(nf));
  my(target = conj(gamma));
  my(best = 1);
  my(besterr = abs(eval_aut_at_complex_root(auts[1], nf, gamma) - target));
  my(val, err);

  for(i = 2, #auts,
    val = eval_aut_at_complex_root(auts[i], nf, gamma);
    err = abs(val - target);

    if(err < besterr,
      besterr = err;
      best = i;
    );
  );

  if(besterr > 10^(-40), error("Numerical identification of complex conjugation failed; increase realprecision or check the field."));

  return(auts[best]);
}


tau = find_complex_conjugation_from_auts(Labs, all_galois_automorphisms);


/* ===== Downward map from ideals of L to ideals of E ===== */

/*
  Convert the absolute prime ideal to a Z-basis before passing it to
  rnfidealabstorel. This avoids inconsistent-dimension errors.
*/

ideal_down_to_E_hnf(rnf, Enf, Labs, pr_abs) =
{
  my(pr_hnf = idealhnf(Labs, pr_abs));
  my(pr_basis = Labs.zk * pr_hnf);
  my(pr_rel = rnfidealabstorel(rnf, pr_basis));
  my(pr_down = rnfidealdown(rnf, pr_rel));

  idealhnf(Enf, pr_down);
}


/*
  Find index j such that aut(dec[i]) = dec[j].
  This is only used to identify the conjugate side at setup time.
*/

find_image_prime_index(nf, dec, aut, i) =
{
  my(img = nfgaloisapply(nf, aut, dec[i]));

  for(j = 1, #dec,
    if(idealval(nf, img, dec[j]) == 1,
      return(j);
    );
  );

  error("Could not identify image of prime ideal under automorphism");
}


/* ===== Distinguish original and conjugate primes above 2 ===== */

/*
  Let dec[1] be P1.

  original_set_of_primes consists of all dec[i] lying above the same
  prime of E as dec[1].

  conjugate_set_of_primes is obtained by applying complex conjugation.
*/

P1_down = ideal_down_to_E_hnf(rnf_LE, Enf, Labs, dec[1]);

original_set_of_primes = List();

{
for(i = 1, #dec,
  if(ideal_down_to_E_hnf(rnf_LE, Enf, Labs, dec[i]) == P1_down,
    listput(original_set_of_primes, i);
  );
);
}

original_set_of_primes = Vec(original_set_of_primes);
s = #original_set_of_primes;


conjugate_set_of_primes = List();

{
for(i = 1, s,
  j = find_image_prime_index(Labs, dec, tau, original_set_of_primes[i]);
  listput(conjugate_set_of_primes, j);
);
}

conjugate_set_of_primes = Vec(conjugate_set_of_primes);


/*
  Sanity checks.

  Here #dec = 2*s should hold because original_set_of_primes and
  conjugate_set_of_primes are the two conjugate sides.
*/

if(#dec != 2*s, error("#dec is not equal to 2*s; conjugate decomposition is invalid."));
if(#Set(concat(original_set_of_primes, conjugate_set_of_primes)) != #dec, error("original_set_of_primes and conjugate_set_of_primes do not form a disjoint partition of dec."));


/* ===== Class-group logarithm functions ===== */

class_group_cyc = Lbnf.cyc;
number_of_class_group_components = #class_group_cyc;

reduce_class_log(v) =
{
  my(r = #class_group_cyc);
  vector(r, i, Mod(v[i], class_group_cyc[i]));
}


class_log_add(u, v) =
{
  my(r = #class_group_cyc);
  vector(r, i, u[i] + v[i]);
}


class_log_scalar_mul(a, v) =
{
  my(r = #class_group_cyc);
  vector(r, i, a * v[i]);
}


zero_class_log = vector(number_of_class_group_components, i, Mod(0, class_group_cyc[i]));


/*
  Compute full class-group logarithms of the relevant prime ideals.
*/

discrete_log_of_original = List();

{
for(i = 1, s,
  discrete_log = bnfisprincipal(Lbnf, dec[original_set_of_primes[i]], 0);
  listput(discrete_log_of_original, Vec(discrete_log));
);
}

discrete_log_of_conjugate = List();

{
for(i = 1, s,
  discrete_log = bnfisprincipal(Lbnf, dec[conjugate_set_of_primes[i]], 0);
  listput(discrete_log_of_conjugate, Vec(discrete_log));
);
}

discrete_log_of_original = Vec(discrete_log_of_original);
discrete_log_of_conjugate = Vec(discrete_log_of_conjugate);


mod_discrete_log_of_original = List();

{
for(i = 1, s,
  mod_discrete_log = reduce_class_log(discrete_log_of_original[i]);
  listput(mod_discrete_log_of_original, mod_discrete_log);
);
}

mod_discrete_log_of_original = Vec(mod_discrete_log_of_original);


mod_discrete_log_of_conjugate = List();

{
for(i = 1, s,
  mod_discrete_log = reduce_class_log(discrete_log_of_conjugate[i]);
  listput(mod_discrete_log_of_conjugate, mod_discrete_log);
);
}

mod_discrete_log_of_conjugate = Vec(mod_discrete_log_of_conjugate);


/* ===== Bounded partitions ===== */

global_solutions = List();

partition_worker(n, idx, rem, max_val, vec) =
{
  if(idx == n,
    if(rem >= 0 && rem <= max_val,
      vec[n] = rem;
      listput(global_solutions, vec);
    );
    return();
  );

  my(limit = min(max_val, rem));
  my(remaining_parts = n - idx + 1);
  my(min_val = (rem + remaining_parts - 1) \ remaining_parts);

  forstep(val = limit, min_val, -1,
    vec[idx] = val;
    partition_worker(n, idx + 1, rem - val, val, vec);
  );
}


get_bounded_partitions(n, m, k) =
{
  global_solutions = List();

  if(n <= 0, return([]));
  if(m < 0, return([]));
  if(m > n*k, return([]));

  my(current_vec = vector(n));

  partition_worker(n, 1, m, k, current_vec);

  return(Vec(global_solutions));
}


/* ===== Distinct permutations ===== */

get_distinct_permutations(L) =
{
  my(v = vecsort(L));
  my(n = #v);
  my(result = List());

  listput(result, v);

  if(n <= 1,
    return(Vec(result));
  );

  while(1,
    my(i = n - 1);

    while(i > 0 && v[i] >= v[i+1], i--);

    if(i == 0, break);

    my(j = n);

    while(v[j] <= v[i], j--);

    my(temp = v[i]);
    v[i] = v[j];
    v[j] = temp;

    my(left = i + 1);
    my(right = n);

    while(left < right,
      temp = v[left];
      v[left] = v[right];
      v[right] = temp;
      left++;
      right--;
    );

    listput(result, v);
  );

  return(Vec(result));
}


/* ===== Norm-to-E pruning ===== */

solve_constrained_sum(total_sum, divisor) =
{
  my(solutions = List());
  my(start_k = (total_sum \ divisor) * divisor);

  forstep(k = start_k, 0, -divisor,
    if((total_sum - k) % 2 == 0,
      my(A = (total_sum + k) / 2);
      my(B = (total_sum - k) / 2);
      listput(solutions, [A, B]);
    );
  );

  return(Vec(solutions));
}


/* ===== Convert a conjugate-side vector to a full exponent vector ===== */

full_vector_from_conjugate(bl, conjugate_ideal) =
{
  my(PIP_ideal = vector(#dec, k, 0));

  for(j = 1, s,
    PIP_ideal[conjugate_set_of_primes[j]] = conjugate_ideal[j];
    PIP_ideal[original_set_of_primes[j]] = bl - conjugate_ideal[j];
  );

  return(PIP_ideal);
}


/* ===== Recover exponent vector of an ideal at the primes in dec ===== */

valuation_vector_at_dec(I) =
{
  vector(#dec, k, idealval(Labs, I, dec[k]));
}


/* ===== Apply all automorphisms to one full exponent vector ===== */

recover_galois_orbit_vectors(full_vec) =
{
  my(res = List());
  my(I = idealfactorback(Labs, dec, full_vec));
  my(J, exp_vec);

  for(a = 1, #all_galois_automorphisms,
    J = nfgaloisapply(Labs, all_galois_automorphisms[a], I);
    exp_vec = valuation_vector_at_dec(J);
    listput(res, exp_vec);
  );

  return(Vec(res));
}


/* ===== Solve the PIP for one odd exponent bl ===== */

/*
  Return value:

      [reduced_conjugate_solutions,
       full_solution_to_PIP,
       support_of_solutions,
       list_of_same_support]

  The key correction is inside the recovery stage:

      representative full ideal
      -> apply every automorphism in Gal(L/Q)
      -> recover exponents via idealval
      -> remove duplicates.

  We do NOT use allshifts.
*/

solve_PIP_for_length(bl) =
{
  my(cases_to_be_checked = solve_constrained_sum(s * bl, norm_pruning_divisor));
  my(L_construct = vector(s, i, bl));
  my(collection_of_solutions_to_PIP = List());

  for(case = 1, #cases_to_be_checked,

    my(amount_of_conjugate_ideals = cases_to_be_checked[case][2]);
    my(partition_of_conjugate_ideals = get_bounded_partitions(s, amount_of_conjugate_ideals, bl));

    for(i = 1, #partition_of_conjugate_ideals,

      my(list_type_of_conjugate = List());

      if(s == 1,
        listput(list_type_of_conjugate, [partition_of_conjugate_ideals[i][1]]),
        my(type_of_conjugate = partition_of_conjugate_ideals[i][2..s]);
        my(list_type_of_conjugate_construct = get_distinct_permutations(type_of_conjugate));

        for(k = 1, #list_type_of_conjugate_construct,
          listput(list_type_of_conjugate, concat(partition_of_conjugate_ideals[i][1], list_type_of_conjugate_construct[k]));
        );
      );

      my(list_type_of_original = List());

      for(j = 1, #list_type_of_conjugate,
        listput(list_type_of_original, L_construct - list_type_of_conjugate[j]);
      );

      for(j = 1, #list_type_of_conjugate,

        my(conjugate_ideal = list_type_of_conjugate[j]);
        my(original_ideal = list_type_of_original[j]);

        my(count_discrete_log = zero_class_log);

        for(k = 1, s,
          my(add_discrete_log = class_log_scalar_mul(conjugate_ideal[k], mod_discrete_log_of_conjugate[k]));
          count_discrete_log = class_log_add(count_discrete_log, add_discrete_log);
        );

        for(k = 1, s,
          my(add_discrete_log = class_log_scalar_mul(original_ideal[k], mod_discrete_log_of_original[k]));
          count_discrete_log = class_log_add(count_discrete_log, add_discrete_log);
        );

        if(count_discrete_log == zero_class_log,
          listput(collection_of_solutions_to_PIP, conjugate_ideal);
        );
      );
    );
  );


  /*
    Correct recovery of the full solution list.

    Each element of collection_of_solutions_to_PIP gives one full ideal.
    Then we apply every element of Gal(L/Q) and recover valuations at dec.
  */

  my(full_solution_to_PIP_list = List());

  for(i = 1, #collection_of_solutions_to_PIP,

    my(rep_full_vec = full_vector_from_conjugate(bl, collection_of_solutions_to_PIP[i]));
    my(orbit_vecs = recover_galois_orbit_vectors(rep_full_vec));

    for(j = 1, #orbit_vecs,
      listput(full_solution_to_PIP_list, orbit_vecs[j]);
    );
  );

  my(full_solution_to_PIP = Set(Vec(full_solution_to_PIP_list)));


  /*
    Same-support check.

    We record the zero-supports, i.e. complements of supports among
    primes above 2. Equality of zero-supports is equivalent to equality
    of supports.
  */

  my(support_of_solutions = List());

  for(i = 1, #full_solution_to_PIP,
    listput(support_of_solutions, select(x -> x == 0, full_solution_to_PIP[i], 1));
  );

  my(list_of_same_support = List());

  for(i = 1, #support_of_solutions - 1,
    for(j = i + 1, #support_of_solutions,
      if(support_of_solutions[i] == support_of_solutions[j],
        listput(list_of_same_support, [i, j]);
      );
    );
  );

  return([Vec(collection_of_solutions_to_PIP), full_solution_to_PIP, Vec(support_of_solutions), Vec(list_of_same_support)]);
}


/* ===== Optional principalness checker for debugging ===== */

/*
  This should return [] if the recovered full_solution_to_PIP list is correct.
*/

nonprincipal_full_solution_indices(full_solution_to_PIP) =
{
  my(bad = List());

  for(i = 1, #full_solution_to_PIP,
    my(I = idealfactorback(Labs, dec, full_solution_to_PIP[i]));
    my(v = bnfisprincipal(Lbnf, I, 0));

    if(Vec(v) != vector(#class_group_cyc, k, 0),
      listput(bad, i);
    );
  );

  return(Vec(bad));
}


/* ===== Main loop for the element partition method ===== */

first_solution_n = -1;
first_result = [];

{
forstep(current_n = start_n, max_n, 2,

  print("Checking n = ", current_n, " ...");

  result = solve_PIP_for_length(current_n);

  reduced_conjugate_solutions = result[1];
  full_solution_to_PIP = result[2];
  support_of_solutions = result[3];
  list_of_same_support = result[4];

  print("  number of reduced conjugate-part solutions = ", #reduced_conjugate_solutions);
  print("  number of full PIP solutions = ", #full_solution_to_PIP);

  if(#full_solution_to_PIP == 0,
    next();
  );

  first_solution_n = current_n;
  first_result = result;

  print("First n with PIP solutions = ", first_solution_n);
  print("number of reduced conjugate-part solutions = ", #reduced_conjugate_solutions);
  print("number of full solutions to PIP = ", #full_solution_to_PIP);

  bad_principal_indices = nonprincipal_full_solution_indices(full_solution_to_PIP);

  if(#bad_principal_indices == 0,
    print("Principalness verification PASSED for all recovered full solutions.");
  ,
    error("Principalness verification FAILED for at least one recovered full solution.");
  );

  if(#list_of_same_support == 0,
    print("Same-support check PASSED.");
    print("Conclusion: by the element partition method, there is no GBF of type [t, 2N] for odd t <= ", first_solution_n, ".");
    print("The method cannot do better beyond this point, since a solution at n gives a solution divisible by 2 at n+2.");
  ,
    print("Same-support check FAILED.");
    print("Pairs with the same support: ", list_of_same_support);
    print("The method proves nonexistence only for odd t < ", first_solution_n, " by the no-solution check.");
  );

  print("full_solution_to_PIP = ", full_solution_to_PIP);

  break;
);
}

if(first_solution_n == -1, print("No PIP solution was found for odd n in the range ", start_n, " <= n <= ", max_n));


/* ===== Final basic data ===== */

print("p1 = ", p1);
print("p2 = ", p2);
print("N = ", N);
print("m_E = ", m_E);
print("ord_2_in_E = ", ord_2_in_E);
print("norm_pruning_divisor = ", norm_pruning_divisor);
print("expected degree of L = ", degL_expected);
print("degree of L = ", poldegree(Labs.pol));
print("number of prime ideals above 2 in L = ", #dec);
print("2 * number of prime ideals above 2 in L = ", 2 * #dec);
print("s = ", s);
print("original_set_of_primes = ", original_set_of_primes);
print("conjugate_set_of_primes = ", conjugate_set_of_primes);
print("class_group_cyc = ", class_group_cyc);
