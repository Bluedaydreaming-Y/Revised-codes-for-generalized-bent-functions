/*
  WARNING:
  This computation uses bnfinit without bnfcertify.
  Therefore the output should be regarded as conditional on GRH,
  exactly as stated in Section 6 of the paper.
*/

/*
  Section 6, Case (III): norm-pruned element-partition-method search for

      p1 == 3 mod 8,
      p2 == 1 mod 8,
      kronecker(p1,p2) = -1,
      N = p1 * p2.

  We solve the PIP in the decomposition field D of 2 in K = Q(zeta_N):

      I * conjugate(I) = 2^n O_D.

  Norm-pruning:

      Let T be the unique quartic subfield of D.
      Let [D:T] = s, so [D:Q] = 4s.

      Let 2 O_T = p0 p1 p2 p3.
      Let S_i be the set of prime ideals of O_D above p_i.

      For a candidate I, let r_i be the total exponent of primes in S_i.
      Then

          r_0 + r_2 = n*s,
          r_1 + r_3 = n*s,

      and

          N_{D/T}(I) = p0^r0 p1^r1 p2^r2 p3^r3

      must be principal.

  Important:

      The code computes complex conjugation directly and uses tau_partner.
      It does not assume that complex conjugation is sigma^2.

      This is the cyclic version. Therefore the program stops with an error
      if it cannot find an automorphism acting transitively on the primes
      above 2.
*/


/* ===== Input ===== */

p1 = 67;
p2 = 97;

start_n = 3;
max_n = 101;

default(realprecision, 100);


/* ===== Basic validation ===== */

if(!isprime(p1), error("p1 must be prime"));
if(!isprime(p2), error("p2 must be prime"));
if(p1 == p2, error("p1 and p2 must be distinct"));

if(p1 % 8 != 3, error("p1 must be congruent to 3 modulo 8"));
if(p2 % 8 != 1, error("p2 must be congruent to 1 modulo 8"));

if(kronecker(p1, p2) != -1, error("kronecker(p1,p2) must be -1"));

if(start_n < 1, error("start_n must be positive"));
if(start_n % 2 == 0, error("start_n must be odd"));
if(max_n < start_n, error("max_n must be at least start_n"));


/* ===== Check the p2 order / non-quartic-residue condition ===== */

/*
  Since p2 == 1 mod 8, write

      p2 - 1 = 2^lambda_p2 * odd.

  The required non-quartic-residue/order condition is

      valuation(ord_p2(2), 2) = lambda_p2 - 1.

  Equivalently, 2 is a quadratic residue but not a quartic residue modulo p2.
*/

lambda_p2 = valuation(p2 - 1, 2);
f2_order = znorder(Mod(2, p2));
v2_f2 = valuation(f2_order, 2);

if(lambda_p2 < 3, error("lambda_p2 must be at least 3 because p2 should be congruent to 1 modulo 8"));
if(v2_f2 != lambda_p2 - 1, error("The required p2 order condition fails: valuation(ord_p2(2),2) must equal valuation(p2-1,2)-1"));


/* ===== Utility: change polynomial variable ===== */

change_variable(pol, newvar) =
{
  return(subst(pol, variable(pol), newvar));
}


/* ===== Find quartic subfield polynomial ===== */

find_quartic_subfield_pol(N) =
{
  my(fT);

  for(a = 2, N - 1,
    if(gcd(a, N) == 1,
      fT = galoissubcyclo(N, [2, a]);

      if(poldegree(fT) == 4,
        return(fT);
      );
    );
  );

  error("Could not find the quartic subfield T using galoissubcyclo(N,[2,a])");
}


/* ===== Find relative factor of D over T ===== */

find_relative_factor(Tnf, fD_abs, expected_degree) =
{
  my(fac = nffactor(Tnf, fD_abs));
  my(pol);

  for(i = 1, matsize(fac)[1],
    pol = fac[i, 1];

    if(poldegree(pol) == expected_degree,
      return(pol);
    );
  );

  error("Could not find a relative defining polynomial of the expected degree");
}


/* ===== Automorphism helpers ===== */

eval_aut_at_complex_root(aut, nf, gamma) =
{
  my(v = variable(nf.pol));
  subst(lift(aut), v, gamma);
}


choose_nonreal_root(nf) =
{
  my(rr = nf.roots);

  for(i = 1, #rr,
    if(abs(imag(rr[i])) > 10^(-30),
      return(rr[i]);
    );
  );

  error("No numerically nonreal root found. Increase realprecision or check the field");
}


find_complex_conjugation_from_auts(nf, auts) =
{
  my(gamma = choose_nonreal_root(nf));
  my(target = conj(gamma));
  my(best = 1);
  my(besterr = abs(eval_aut_at_complex_root(auts[1], nf, gamma) - target));
  my(val, err);

  for(i = 2, #auts,
    val = eval_aut_at_complex_root(auts[i], nf, gamma);
    err = abs(val - target);

    if(err < besterr,
      besterr = err;
      best = i;
    );
  );

  if(besterr > 10^(-40), error("Numerical identification of complex conjugation failed; increase realprecision or check the field"));

  return(auts[best]);
}


find_image_prime_index(nf, dec, aut, i) =
{
  my(img = nfgaloisapply(nf, aut, dec[i]));

  for(j = 1, #dec,
    if(idealval(nf, img, dec[j]) == 1,
      return(j);
    );
  );

  error("Could not identify image of prime ideal under automorphism");
}


orbit_of_dec1_under_aut(nf, dec, aut) =
{
  my(n = #dec);
  my(orb = vector(n, k, 0));
  my(idx = 1);

  for(k = 1, n,
    if(vecsearch(orb, idx),
      return([]);
    );

    orb[k] = idx;
    idx = find_image_prime_index(nf, dec, aut, idx);
  );

  if(idx != 1,
    return([]);
  );

  return(orb);
}


find_prime_cycle_generator(nf, dec, auts) =
{
  my(orb);

  for(i = 1, #auts,
    orb = orbit_of_dec1_under_aut(nf, dec, auts[i]);

    if(#orb == #dec,
      return([auts[i], orb]);
    );
  );

  error("Could not find an automorphism acting transitively on the primes above 2. This cyclic program is not applicable.");
}


/* ===== Downward map from ideals of D to ideals of T ===== */

ideal_down_to_T_hnf(rnf, Tnf, Fabs, pr_abs) =
{
  my(pr_hnf = idealhnf(Fabs, pr_abs));
  my(pr_basis = Fabs.zk * pr_hnf);
  my(pr_rel = rnfidealabstorel(rnf, pr_basis));
  my(pr_down = rnfidealdown(rnf, pr_rel));

  idealhnf(Tnf, pr_down);
}


/* ===== Class-group vector helpers ===== */

reduce_class_log(v) =
{
  my(r = #class_group_cyc);
  vector(r, i, Mod(v[i], class_group_cyc[i]));
}


class_log_add(u, v) =
{
  my(r = #class_group_cyc);
  vector(r, i, u[i] + v[i]);
}


class_log_scalar_mul(a, v) =
{
  my(r = #class_group_cyc);
  vector(r, i, a * v[i]);
}


class_log_of_full_vector(full_vec) =
{
  my(count_discrete_log = zero_class_log);

  for(k = 1, #dec,
    my(add_discrete_log = class_log_scalar_mul(full_vec[k], mod_discrete_log_of_dec[k]));
    count_discrete_log = class_log_add(count_discrete_log, add_discrete_log);
  );

  return(count_discrete_log);
}


is_principal_full_vector(full_vec) =
{
  return(class_log_of_full_vector(full_vec) == zero_class_log);
}


/* ===== Principalness in T for norm pruning ===== */

is_principal_T_exponent_vector(rvec) =
{
  my(J = idealfactorback(Tnf, pT, rvec));
  my(v = bnfisprincipal(Tbnf, J, 0));
  my(zero_T = vector(#Tbnf.cyc, k, 0));

  return(Vec(v) == zero_T);
}


valid_norm_pairs_for_length(bl) =
{
  my(valid = List());
  my(ns = bl * s_ext);
  my(r0, r1, r2, r3);

  for(r0 = 0, ns,
    for(r1 = 0, ns,
      r2 = ns - r0;
      r3 = ns - r1;

      if(is_principal_T_exponent_vector([r0, r1, r2, r3]),
        listput(valid, [r0, r1]);
      );
    );
  );

  return(Vec(valid));
}


/* ===== Bounded partitions and permutations ===== */

global_solutions = List();

partition_worker(n, idx, rem, max_val, vec) =
{
  if(idx == n,
    if(rem >= 0 && rem <= max_val,
      vec[n] = rem;
      listput(global_solutions, vec);
    );
    return();
  );

  my(limit = min(max_val, rem));
  my(remaining_parts = n - idx + 1);
  my(min_val = (rem + remaining_parts - 1) \ remaining_parts);

  forstep(val = limit, min_val, -1,
    vec[idx] = val;
    partition_worker(n, idx + 1, rem - val, val, vec);
  );
}


get_bounded_partitions(n, m, k) =
{
  global_solutions = List();

  if(n <= 0, return([]));
  if(m < 0, return([]));
  if(m > n*k, return([]));

  my(current_vec = vector(n));

  partition_worker(n, 1, m, k, current_vec);

  return(Vec(global_solutions));
}


get_distinct_permutations(L) =
{
  my(v = vecsort(L));
  my(n = #v);
  my(result = List());

  listput(result, v);

  if(n <= 1,
    return(Vec(result));
  );

  while(1,
    my(i = n - 1);

    while(i > 0 && v[i] >= v[i+1], i--);

    if(i == 0,
      break;
    );

    my(j = n);

    while(v[j] <= v[i], j--);

    my(temp = v[i]);
    v[i] = v[j];
    v[j] = temp;

    my(left = i + 1);
    my(right = n);

    while(left < right,
      temp = v[left];
      v[left] = v[right];
      v[right] = temp;
      left++;
      right--;
    );

    listput(result, v);
  );

  return(Vec(result));
}


vectors_for_S0_sum(total, bl) =
{
  my(parts = get_bounded_partitions(s_ext, total, bl));
  my(result = List());

  for(i = 1, #parts,
    if(s_ext == 1,
      listput(result, [parts[i][1]]);
    ,
      my(tail = parts[i][2..s_ext]);
      my(tail_perms = get_distinct_permutations(tail));

      for(k = 1, #tail_perms,
        listput(result, concat(parts[i][1], tail_perms[k]));
      );
    );
  );

  return(Set(Vec(result)));
}


all_bounded_vectors_for_sum(total, bl) =
{
  my(parts = get_bounded_partitions(s_ext, total, bl));
  my(result = List());

  for(i = 1, #parts,
    my(perms = get_distinct_permutations(parts[i]));

    for(k = 1, #perms,
      listput(result, perms[k]);
    );
  );

  return(Set(Vec(result)));
}


/* ===== Build full vector from S0 and S1 exponent data ===== */

full_vector_from_S0_S1(bl, v0, v1) =
{
  my(full_vec = vector(#dec, k, 0));
  my(idx);

  for(j = 1, s_ext,
    idx = S0_indices[j];
    full_vec[idx] = v0[j];
    full_vec[tau_partner[idx]] = bl - v0[j];

    idx = S1_indices[j];
    full_vec[idx] = v1[j];
    full_vec[tau_partner[idx]] = bl - v1[j];
  );

  return(full_vec);
}


/* ===== Recover exponent vector of an ideal at dec ===== */

valuation_vector_at_dec(I) =
{
  vector(#dec, k, idealval(Dabs, I, dec[k]));
}


/* ===== Apply all automorphisms to one full exponent vector ===== */

recover_galois_orbit_vectors(full_vec) =
{
  my(res = List());
  my(I = idealfactorback(Dabs, dec, full_vec));
  my(J, exp_vec);

  for(a = 1, #all_galois_automorphisms,
    J = nfgaloisapply(Dabs, all_galois_automorphisms[a], I);
    exp_vec = valuation_vector_at_dec(J);
    listput(res, exp_vec);
  );

  return(Vec(res));
}


/* ===== Solve PIP for one odd exponent bl ===== */

solve_PIP_for_length(bl) =
{
  my(valid_pairs = valid_norm_pairs_for_length(bl));
  my(representative_full_solutions = List());

  print("  number of norm-pruned pairs (r0,r1) = ", #valid_pairs);

  for(pair_index = 1, #valid_pairs,
    my(r0 = valid_pairs[pair_index][1]);
    my(r1 = valid_pairs[pair_index][2]);

    my(S0_vecs = vectors_for_S0_sum(r0, bl));
    my(S1_vecs = all_bounded_vectors_for_sum(r1, bl));

    for(i = 1, #S0_vecs,
      for(j = 1, #S1_vecs,
        my(full_vec = full_vector_from_S0_S1(bl, S0_vecs[i], S1_vecs[j]));

        if(is_principal_full_vector(full_vec),
          listput(representative_full_solutions, full_vec);
        );
      );
    );
  );

  representative_full_solutions = Set(Vec(representative_full_solutions));

  my(full_solution_to_PIP_list = List());

  for(i = 1, #representative_full_solutions,
    my(orbit_vecs = recover_galois_orbit_vectors(representative_full_solutions[i]));

    for(j = 1, #orbit_vecs,
      listput(full_solution_to_PIP_list, orbit_vecs[j]);
    );
  );

  my(full_solution_to_PIP = Set(Vec(full_solution_to_PIP_list)));

  my(support_of_solutions = List());

  for(i = 1, #full_solution_to_PIP,
    listput(support_of_solutions, select(x -> x == 0, full_solution_to_PIP[i], 1));
  );

  my(list_of_same_support = List());

  for(i = 1, #support_of_solutions - 1,
    for(j = i + 1, #support_of_solutions,
      if(support_of_solutions[i] == support_of_solutions[j],
        listput(list_of_same_support, [i, j]);
      );
    );
  );

  return([representative_full_solutions, full_solution_to_PIP, Vec(support_of_solutions), Vec(list_of_same_support)]);
}


/* ===== Optional principalness checker ===== */

nonprincipal_full_solution_indices(full_solution_to_PIP) =
{
  my(bad = List());

  for(i = 1, #full_solution_to_PIP,
    my(I = idealfactorback(Dabs, dec, full_solution_to_PIP[i]));
    my(v = bnfisprincipal(Dbnf, I, 0));

    if(Vec(v) != vector(#class_group_cyc, k, 0),
      listput(bad, i);
    );
  );

  return(Vec(bad));
}


/* ===== Field construction: case s = 1 ===== */

setup_fields_s1() =
{
  fT0 = change_variable(fD0, y);

  Tnf = nfinit(fT0);
  Tbnf = bnfinit(Tnf);

  Dabs = Tnf;
  Dbnf = Tbnf;

  dec = idealprimedec(Dabs, 2);

  D_equals_T = 1;
}


/* ===== Field construction: case s > 1 ===== */

setup_fields_sgt1() =
{
  fT0 = find_quartic_subfield_pol(N);
  fT0 = change_variable(fT0, y);

  Tnf = nfinit(fT0);
  Tbnf = bnfinit(Tnf);

  relpol_DT = find_relative_factor(Tnf, fD0, s_ext);
  rnf_DT = rnfinit(Tnf, relpol_DT, 1);

  Dabs = nfinit(rnf_DT);
  Dbnf = bnfinit(Dabs);

  dec = idealprimedec(Dabs, 2);

  D_equals_T = 0;
}


/* ===== Initialize fields and prime data ===== */

initialize_all() =
{
  if(!isprime(p1), error("p1 must be prime"));
  if(!isprime(p2), error("p2 must be prime"));
  if(p1 == p2, error("p1 and p2 must be distinct"));

  if(p1 % 8 != 3, error("p1 must be congruent to 3 modulo 8"));
  if(p2 % 8 != 1, error("p2 must be congruent to 1 modulo 8"));

  if(kronecker(p1, p2) != -1, error("kronecker(p1,p2) must be -1"));

  if(start_n < 1, error("start_n must be positive"));
  if(start_n % 2 == 0, error("start_n must be odd"));
  if(max_n < start_n, error("max_n must be at least start_n"));

  lambda_p2 = valuation(p2 - 1, 2);
  f2_order = znorder(Mod(2, p2));
  v2_f2 = valuation(f2_order, 2);

  if(lambda_p2 < 3, error("lambda_p2 must be at least 3 because p2 should be congruent to 1 modulo 8"));
  if(v2_f2 != lambda_p2 - 1, error("The required p2 order condition fails: valuation(ord_p2(2),2) must equal valuation(p2-1,2)-1"));

  N = p1 * p2;

  f_N = znorder(Mod(2, N));
  phi_N = eulerphi(N);

  if(phi_N % f_N != 0, error("Unexpected: ord_N(2) does not divide phi(N)"));

  degD_expected = phi_N / f_N;

  if(degD_expected % 4 != 0, error("[D:Q] is not divisible by 4"));

  s_ext = degD_expected / 4;

  if(s_ext % 2 == 0, error("s = [D:T] is not odd; this program assumes the Section 6 Case (III) parameter setting"));

  if(degD_expected > 30, print("Warning: [D:Q] > 30. bnfinit may be very slow."));
  if(s_ext > 7, print("Warning: s > 7. This program may be very slow."));

  print("N = ", N);
  print("ord_N(2) = ", f_N);
  print("phi(N) = ", phi_N);
  print("expected [D:Q] = ", degD_expected);
  print("s = [D:T] = ", s_ext);
  print("lambda_p2 = ", lambda_p2);
  print("ord_p2(2) = ", f2_order);
  print("valuation(ord_p2(2),2) = ", v2_f2);

  fD0 = galoissubcyclo(N, 2);

  if(poldegree(fD0) != degD_expected, error("Degree of fD0 is not phi(N)/ord_N(2)"));

  if(s_ext == 1,
    setup_fields_s1();
  ,
    setup_fields_sgt1();
  );

  if(poldegree(Dabs.pol) != degD_expected, error("The constructed field D has the wrong degree"));
  if(#dec != degD_expected, error("2 does not split completely in D: #dec is not equal to expected [D:Q]"));

  G_all = galoisinit(Dabs);
  all_galois_automorphisms = Vec(galoispermtopol(G_all, G_all.group));

  if(#all_galois_automorphisms != poldegree(Dabs.pol), error("The number of automorphisms is not equal to [D:Q]; D/Q does not appear to be Galois as required"));

  sigma_data = find_prime_cycle_generator(Dabs, dec, all_galois_automorphisms);
  sigma_aut = sigma_data[1];
  P_orbit = sigma_data[2];

  tau = find_complex_conjugation_from_auts(Dabs, all_galois_automorphisms);

  tau_partner = vector(#dec, i, 0);

  for(i = 1, #dec,
    tau_partner[i] = find_image_prime_index(Dabs, dec, tau, i);
  );

  for(i = 1, #dec,
    if(tau_partner[i] == i,
      error("A prime above 2 is fixed by complex conjugation; pairing strategy invalid")
    );

    if(tau_partner[tau_partner[i]] != i,
      error("Complex conjugation pairing is not involutive on dec")
    );
  );

  S0_indices = vector(s_ext, k, P_orbit[1 + 4*(k - 1)]);
  S1_indices = vector(s_ext, k, P_orbit[2 + 4*(k - 1)]);

  if(S0_indices[1] != 1, error("P_0 is not dec[1] as expected"));

  representative_indices = concat(S0_indices, S1_indices);
  conjugate_indices = vector(2*s_ext, k, tau_partner[representative_indices[k]]);

  if(#Set(concat(representative_indices, conjugate_indices)) != #dec, error("representative_indices and conjugate_indices do not partition dec"));

  pT = vector(4, i, 0);

  if(D_equals_T,
    for(i = 1, 4,
      pT[i] = idealhnf(Tnf, dec[P_orbit[i]]);
    );
  ,
    for(i = 1, 4,
      pT[i] = ideal_down_to_T_hnf(rnf_DT, Tnf, Dabs, dec[P_orbit[i]]);
    );
  );

  class_group_cyc = Dbnf.cyc;
  number_of_class_group_components = #class_group_cyc;
  zero_class_log = vector(number_of_class_group_components, i, Mod(0, class_group_cyc[i]));

  mod_discrete_log_of_dec = List();

  for(i = 1, #dec,
    discrete_log = bnfisprincipal(Dbnf, dec[i], 0);
    listput(mod_discrete_log_of_dec, reduce_class_log(Vec(discrete_log)));
  );

  mod_discrete_log_of_dec = Vec(mod_discrete_log_of_dec);
}


/* ===== Main loop ===== */

run_element_partition_loop() =
{
  first_solution_n = -1;
  first_result = [];

  forstep(current_n = start_n, max_n, 2,
    print("Checking n = ", current_n, " ...");

    result = solve_PIP_for_length(current_n);

    representative_full_solutions = result[1];
    full_solution_to_PIP = result[2];
    support_of_solutions = result[3];
    list_of_same_support = result[4];

    print("  number of full PIP solutions = ", #full_solution_to_PIP);

    if(#full_solution_to_PIP == 0,
      next();
    );

    first_solution_n = current_n;
    first_result = result;

    print("First n with PIP solutions = ", first_solution_n);
    print("number of full solutions to PIP = ", #full_solution_to_PIP);

    bad_principal_indices = nonprincipal_full_solution_indices(full_solution_to_PIP);

    if(#bad_principal_indices == 0,
      print("Principalness verification PASSED for all recovered full solutions.");
    ,
      error("Principalness verification FAILED for at least one recovered full solution");
    );

    if(#list_of_same_support == 0,
      print("Same-support check PASSED.");
      print("Conclusion: by the element partition method, there is no GBF of type [t, 2N] for odd t <= ", first_solution_n, ".");
      print("The method cannot do better beyond this point, since a solution at n gives a solution divisible by 2 at n+2.");
    ,
      print("Same-support check FAILED.");
      print("Pairs with the same support: ", list_of_same_support);
      print("The method proves nonexistence only for odd t < ", first_solution_n, " by the no-solution check.");
    );

    print("full_solution_to_PIP = ", full_solution_to_PIP);

    break;
  );

  if(first_solution_n == -1,
    print("No PIP solution was found for odd n in the range ", start_n, " <= n <= ", max_n)
  );
}


/* ===== Final data ===== */

print_final_data() =
{
  print("p1 = ", p1);
  print("p2 = ", p2);
  print("N = ", N);

  print("ord_N(2) = ", f_N);
  print("phi(N) = ", phi_N);
  print("[D:Q] = ", poldegree(Dabs.pol));
  print("expected [D:Q] = ", degD_expected);
  print("s = [D:T] = ", s_ext);

  print("lambda_p2 = ", lambda_p2);
  print("ord_p2(2) = ", f2_order);
  print("valuation(ord_p2(2),2) = ", v2_f2);

  print("number of prime ideals above 2 in D = ", #dec);
  print("P_orbit = ", P_orbit);
  print("S0_indices = ", S0_indices);
  print("S1_indices = ", S1_indices);
  print("tau_partner = ", tau_partner);

  print("class_group_cyc(D) = ", class_group_cyc);
  print("class_group_cyc(T) = ", Tbnf.cyc);
}


/* ===== Main ===== */

main() =
