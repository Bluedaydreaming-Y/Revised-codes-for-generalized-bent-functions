/*
  WARNING:
  This computation uses bnfinit without bnfcertify.
  Therefore the output should be regarded as conditional on GRH,
  exactly as stated in Section 6 of the paper.
*/

/*
  Section 6, Case (III): element-partition-method search for

      N = p1^e1 * p2^e2,
      p1 == 3 mod 8,
      p2 == 1 mod 8,
      kronecker(p1,p2) = -1.

  K = Q(zeta_N).
  D is the decomposition field of 2 in K.

  We solve the PIP directly in O_D:

      I * conjugate(I) = 2^n O_D.

  Difference from the p1 == 7 mod 8, p2 == 1 mod 8 case:

  D does not contain an imaginary quadratic subfield suitable for
  norm-pruning. Therefore we omit the norm-pruning step and test all
  possible exponent vectors.

  Main strategy:

  1. Compute dec = prime ideals of D above 2.
  2. Find the complex conjugation tau.
  3. Pair each prime ideal with its complex conjugate.
  4. Choose one representative from each conjugate pair, always including dec[1].
  5. For each odd n = 3,5,7,...:
       - enumerate all nonnegative exponent vectors on representatives,
         with each exponent <= n;
       - up to Galois action, fix the largest exponent at dec[1];
       - test principalness;
       - recover all Galois conjugates by applying Gal(D/Q);
       - remove duplicates;
       - perform the same-support check.
*/


/* ===== Input ===== */

p1 = 3;
e1 = 1;

p2 = 41;
e2 = 1;

start_n = 3;
max_n = 101;

default(realprecision, 100);


/* ===== Basic validation ===== */

if(!isprime(p1), error("p1 must be prime"));
if(!isprime(p2), error("p2 must be prime"));
if(p1 == p2, error("p1 and p2 must be distinct"));

if(e1 < 1, error("e1 must be positive"));
if(e2 < 1, error("e2 must be positive"));

if(p1 % 8 != 3, error("p1 must be congruent to 3 modulo 8"));
if(p2 % 8 != 1, error("p2 must be congruent to 1 modulo 8"));

if(kronecker(p1, p2) != -1, error("kronecker(p1,p2) must be -1"));

if(start_n < 1, error("start_n must be positive"));
if(start_n % 2 == 0, error("start_n must be odd"));
if(max_n < start_n, error("max_n must be at least start_n"));

N = p1^e1 * p2^e2;


/* ===== Diagnostic order data ===== */

f_N = znorder(Mod(2, N));
phi_N = eulerphi(N);
g_D = phi_N / f_N;

/*
  In the notation used for this parameter family, one writes
      s_paper = phi(N)/(4*f_N).
  The actual number of prime ideals above 2 in D is g_D.
*/

if(phi_N % f_N != 0, error("Unexpected: f_N does not divide phi_N"));
if(g_D % 2 != 0, error("The number of prime ideals above 2 is odd; conjugate pairing is impossible."));

s_paper = phi_N / (4 * f_N);

ord_p1 = znorder(Mod(2, p1^e1));
ord_p2 = znorder(Mod(2, p2^e2));

print("Diagnostic: ord_N(2) = ", f_N);
print("Diagnostic: phi(N) = ", phi_N);
print("Diagnostic: [D:Q] = phi(N)/ord_N(2) = ", g_D);
print("Diagnostic: s_paper = phi(N)/(4*ord_N(2)) = ", s_paper);


/* ===== Construct D ===== */

/*
  D is the decomposition field of 2 in K = Q(zeta_N).
  The decomposition group is generated by 2, so D is fixed by <2>.
*/

fD0 = galoissubcyclo(N, 2);

Dabs = nfinit(fD0);
Dbnf = bnfinit(Dabs);

dec = idealprimedec(Dabs, 2);

if(poldegree(Dabs.pol) != g_D, error("The constructed decomposition field D has the wrong degree."));
if(#dec != g_D, error("2 does not split completely in D: #dec is not equal to [D:Q]."));


/* ===== Compute all Galois automorphisms of D/Q ===== */

G_all = galoisinit(Dabs);
all_galois_automorphisms = Vec(galoispermtopol(G_all, G_all.group));

if(#all_galois_automorphisms != poldegree(Dabs.pol), error("The number of automorphisms is not equal to [D:Q]; D/Q does not appear to be Galois as required."));


/* ===== Find complex conjugation on D ===== */

/*
  We choose a numerically nonreal root. This function returns the first
  root with visibly nonzero imaginary part. If no such root is found,
  it stops with an error instead of silently using a bad root.
*/

eval_aut_at_complex_root(aut, nf, gamma) =
{
  my(v = variable(nf.pol));
  subst(lift(aut), v, gamma);
}


choose_nonreal_root(nf) =
{
  my(rr = nf.roots);

  for(i = 1, #rr,
    if(abs(imag(rr[i])) > 10^(-30),
      return(rr[i]);
    );
  );

  error("No numerically nonreal root found. Increase realprecision or check whether D is real.");
}


find_complex_conjugation_from_auts(nf, auts) =
{
  my(gamma = choose_nonreal_root(nf));
  my(target = conj(gamma));
  my(best = 1);
  my(besterr = abs(eval_aut_at_complex_root(auts[1], nf, gamma) - target));
  my(val, err);

  for(i = 2, #auts,
    val = eval_aut_at_complex_root(auts[i], nf, gamma);
    err = abs(val - target);

    if(err < besterr,
      besterr = err;
      best = i;
    );
  );

  if(besterr > 10^(-40), error("Numerical identification of complex conjugation failed; increase realprecision or check the field."));

  return(auts[best]);
}


tau = find_complex_conjugation_from_auts(Dabs, all_galois_automorphisms);


/* ===== Image of a prime under an automorphism ===== */

find_image_prime_index(nf, dec, aut, i) =
{
  my(img = nfgaloisapply(nf, aut, dec[i]));

  for(j = 1, #dec,
    if(idealval(nf, img, dec[j]) == 1,
      return(j);
    );
  );

  error("Could not identify image of prime ideal under automorphism");
}


/* ===== Pair prime ideals above 2 by complex conjugation ===== */

tau_partner = vector(#dec, i, 0);

{
for(i = 1, #dec,
  tau_partner[i] = find_image_prime_index(Dabs, dec, tau, i);
);
}

/*
  Sanity checks for conjugate pairing.
*/

{
for(i = 1, #dec,
  if(tau_partner[i] == i,
    error("A prime above 2 is fixed by complex conjugation; the pairing strategy is invalid.")
  );

  if(tau_partner[tau_partner[i]] != i,
    error("Complex conjugation pairing is not involutive on dec.")
  );
);
}


/*
  Choose one representative from each conjugate pair.
  We always include dec[1] as representative.
*/

representative_set_of_primes = List();
paired_marker = vector(#dec, i, 0);

listput(representative_set_of_primes, 1);
paired_marker[1] = 1;
paired_marker[tau_partner[1]] = 1;

{
for(i = 2, #dec,
  if(paired_marker[i] == 0,
    listput(representative_set_of_primes, i);
    paired_marker[i] = 1;
    paired_marker[tau_partner[i]] = 1;
  );
);
}

representative_set_of_primes = Vec(representative_set_of_primes);
num_pairs = #representative_set_of_primes;

conjugate_set_of_primes = vector(num_pairs, i, tau_partner[representative_set_of_primes[i]]);


/*
  Sanity checks.
*/

if(#dec != 2*num_pairs, error("#dec is not equal to 2*num_pairs; conjugate pairing is invalid."));
if(representative_set_of_primes[1] != 1, error("dec[1] was not selected as the first representative"));
if(#Set(concat(representative_set_of_primes, conjugate_set_of_primes)) != #dec, error("representatives and conjugates do not form a disjoint partition of dec."));


/* ===== Class-group logarithm functions ===== */

class_group_cyc = Dbnf.cyc;
number_of_class_group_components = #class_group_cyc;

reduce_class_log(v) =
{
  my(r = #class_group_cyc);
  vector(r, i, Mod(v[i], class_group_cyc[i]));
}


class_log_add(u, v) =
{
  my(r = #class_group_cyc);
  vector(r, i, u[i] + v[i]);
}


class_log_scalar_mul(a, v) =
{
  my(r = #class_group_cyc);
  vector(r, i, a * v[i]);
}


zero_class_log = vector(number_of_class_group_components, i, Mod(0, class_group_cyc[i]));


/*
  Compute class-group logarithms of all prime ideals in dec.
*/

mod_discrete_log_of_dec = List();

{
for(i = 1, #dec,
  discrete_log = bnfisprincipal(Dbnf, dec[i], 0);
  listput(mod_discrete_log_of_dec, reduce_class_log(Vec(discrete_log)));
);
}

mod_discrete_log_of_dec = Vec(mod_discrete_log_of_dec);


/* ===== Principalness test for full exponent vectors ===== */

class_log_of_full_vector(full_vec) =
{
  my(count_discrete_log = zero_class_log);

  for(k = 1, #dec,
    my(add_discrete_log = class_log_scalar_mul(full_vec[k], mod_discrete_log_of_dec[k]));
    count_discrete_log = class_log_add(count_discrete_log, add_discrete_log);
  );

  return(count_discrete_log);
}


is_principal_full_vector(full_vec) =
{
  return(class_log_of_full_vector(full_vec) == zero_class_log);
}


/* ===== Bounded partitions ===== */

/*
  get_bounded_partitions(n,m,k) returns all nonincreasing vectors

      x_1 >= x_2 >= ... >= x_n >= 0

  with sum m and each x_i <= k.

  Note: entries are nonnegative, not positive.
*/

global_solutions = List();

partition_worker(n, idx, rem, max_val, vec) =
{
  if(idx == n,
    if(rem >= 0 && rem <= max_val,
      vec[n] = rem;
      listput(global_solutions, vec);
    );
    return();
  );

  my(limit = min(max_val, rem));
  my(remaining_parts = n - idx + 1);
  my(min_val = (rem + remaining_parts - 1) \ remaining_parts);

  forstep(val = limit, min_val, -1,
    vec[idx] = val;
    partition_worker(n, idx + 1, rem - val, val, vec);
  );
}


get_bounded_partitions(n, m, k) =
{
  global_solutions = List();

  if(n <= 0, return([]));
  if(m < 0, return([]));
  if(m > n*k, return([]));

  my(current_vec = vector(n));

  partition_worker(n, 1, m, k, current_vec);

  return(Vec(global_solutions));
}


/* ===== Distinct permutations ===== */

get_distinct_permutations(L) =
{
  my(v = vecsort(L));
  my(n = #v);
  my(result = List());

  listput(result, v);

  if(n <= 1,
    return(Vec(result));
  );

  while(1,
    my(i = n - 1);

    while(i > 0 && v[i] >= v[i+1], i--);

    if(i == 0, break);

    my(j = n);

    while(v[j] <= v[i], j--);

    my(temp = v[i]);
    v[i] = v[j];
    v[j] = temp;

    my(left = i + 1);
    my(right = n);

    while(left < right,
      temp = v[left];
      v[left] = v[right];
      v[right] = temp;
      left++;
      right--;
    );

    listput(result, v);
  );

  return(Vec(result));
}


/* ===== Convert representative exponents to full exponent vector ===== */

/*
  For each conjugate pair

      representative_set_of_primes[j],
      conjugate_set_of_primes[j],

  the exponents must sum to bl.

  If rep_exp[j] is the exponent on the representative prime, then
  bl - rep_exp[j] is the exponent on the conjugate prime.
*/

full_vector_from_representatives(bl, rep_exp) =
{
  my(PIP_ideal = vector(#dec, k, 0));

  for(j = 1, num_pairs,
    PIP_ideal[representative_set_of_primes[j]] = rep_exp[j];
    PIP_ideal[conjugate_set_of_primes[j]] = bl - rep_exp[j];
  );

  return(PIP_ideal);
}


/* ===== Recover exponent vector of an ideal at the primes in dec ===== */

valuation_vector_at_dec(I) =
{
  vector(#dec, k, idealval(Dabs, I, dec[k]));
}


/* ===== Apply all automorphisms to one full exponent vector ===== */

recover_galois_orbit_vectors(full_vec) =
{
  my(res = List());
  my(I = idealfactorback(Dabs, dec, full_vec));
  my(J, exp_vec);

  for(a = 1, #all_galois_automorphisms,
    J = nfgaloisapply(Dabs, all_galois_automorphisms[a], I);
    exp_vec = valuation_vector_at_dec(J);
    listput(res, exp_vec);
  );

  return(Vec(res));
}


/* ===== Solve the PIP for one odd exponent bl ===== */

/*
  Return value:

      [representative_full_solutions,
       full_solution_to_PIP,
       support_of_solutions,
       list_of_same_support]

  representative_full_solutions are those with the largest representative
  exponent fixed at dec[1].

  full_solution_to_PIP is obtained by applying all elements of Gal(D/Q)
  and then removing duplicates.
*/

solve_PIP_for_length(bl) =
{
  my(representative_full_solutions = List());

  /*
    No norm-pruning here. We test all possible sums of representative
    exponents from 0 to num_pairs*bl.
  */

  for(total_rep_exp = 0, num_pairs * bl,

    my(partitions = get_bounded_partitions(num_pairs, total_rep_exp, bl));

    for(i = 1, #partitions,

      my(list_rep_exp = List());

      /*
        partitions[i] is nonincreasing, so its first entry is maximal.
        We fix this largest entry as the exponent of dec[1].
      */

      if(num_pairs == 1,
        listput(list_rep_exp, [partitions[i][1]]),
        my(tail = partitions[i][2..num_pairs]);
        my(tail_perms = get_distinct_permutations(tail));

        for(k = 1, #tail_perms,
          listput(list_rep_exp, concat(partitions[i][1], tail_perms[k]));
        );
      );

      for(j = 1, #list_rep_exp,

        my(rep_exp = list_rep_exp[j]);
        my(full_vec = full_vector_from_representatives(bl, rep_exp));

        if(is_principal_full_vector(full_vec),
          listput(representative_full_solutions, full_vec);
        );
      );
    );
  );


  /*
    Recover all full solutions by applying every element of Gal(D/Q).
  */

  my(full_solution_to_PIP_list = List());

  for(i = 1, #representative_full_solutions,
    my(orbit_vecs = recover_galois_orbit_vectors(representative_full_solutions[i]));

    for(j = 1, #orbit_vecs,
      listput(full_solution_to_PIP_list, orbit_vecs[j]);
    );
  );

  my(full_solution_to_PIP = Set(Vec(full_solution_to_PIP_list)));


  /*
    Same-support check.

    We record zero-supports, i.e. complements of supports among
    prime ideals above 2. Equality of zero-supports is equivalent to
    equality of supports.
  */

  my(support_of_solutions = List());

  for(i = 1, #full_solution_to_PIP,
    listput(support_of_solutions, select(x -> x == 0, full_solution_to_PIP[i], 1));
  );

  my(list_of_same_support = List());

  for(i = 1, #support_of_solutions - 1,
    for(j = i + 1, #support_of_solutions,
      if(support_of_solutions[i] == support_of_solutions[j],
        listput(list_of_same_support, [i, j]);
      );
    );
  );

  return([Vec(representative_full_solutions), full_solution_to_PIP, Vec(support_of_solutions), Vec(list_of_same_support)]);
}


/* ===== Optional principalness checker for debugging ===== */

/*
  This should return [] if the recovered full_solution_to_PIP list is correct.
*/

nonprincipal_full_solution_indices(full_solution_to_PIP) =
{
  my(bad = List());

  for(i = 1, #full_solution_to_PIP,
    my(I = idealfactorback(Dabs, dec, full_solution_to_PIP[i]));
    my(v = bnfisprincipal(Dbnf, I, 0));

    if(Vec(v) != vector(#class_group_cyc, k, 0),
      listput(bad, i);
    );
  );

  return(Vec(bad));
}


/* ===== Main loop for the element partition method ===== */

first_solution_n = -1;
first_result = [];

{
forstep(current_n = start_n, max_n, 2,

  print("Checking n = ", current_n, " ...");

  result = solve_PIP_for_length(current_n);

  representative_full_solutions = result[1];
  full_solution_to_PIP = result[2];
  support_of_solutions = result[3];
  list_of_same_support = result[4];

  print("  number of full PIP solutions = ", #full_solution_to_PIP);

  if(#full_solution_to_PIP == 0,
    next();
  );

  first_solution_n = current_n;
  first_result = result;

  print("First n with PIP solutions = ", first_solution_n);
  print("number of full solutions to PIP = ", #full_solution_to_PIP);

  bad_principal_indices = nonprincipal_full_solution_indices(full_solution_to_PIP);

  if(#bad_principal_indices == 0,
    print("Principalness verification PASSED for all recovered full solutions.");
  ,
    error("Principalness verification FAILED for at least one recovered full solution.");
  );

  if(#list_of_same_support == 0,
    print("Same-support check PASSED.");
    print("Conclusion: by the element partition method, there is no GBF of type [t, 2N] for odd t <= ", first_solution_n, ".");
    print("The method cannot do better beyond this point, since a solution at n gives a solution divisible by 2 at n+2.");
  ,
    print("Same-support check FAILED.");
    print("Pairs with the same support: ", list_of_same_support);
    print("The method proves nonexistence only for odd t < ", first_solution_n, " by the no-solution check.");
  );

  print("full_solution_to_PIP = ", full_solution_to_PIP);

  break;
);
}

if(first_solution_n == -1, print("No PIP solution was found for odd n in the range ", start_n, " <= n <= ", max_n));


/* ===== Final basic data ===== */

print("p1 = ", p1);
print("e1 = ", e1);
print("p2 = ", p2);
print("e2 = ", e2);
print("N = ", N);

print("ord_p1e1(2) = ", ord_p1);
print("ord_p2e2(2) = ", ord_p2);
print("ord_N(2) = ", f_N);
print("phi(N) = ", phi_N);
print("[D:Q] = ", poldegree(Dabs.pol));
print("expected [D:Q] = phi(N)/ord_N(2) = ", g_D);

print("number of prime ideals above 2 in D = ", #dec);
print("number of conjugate pairs = ", num_pairs);
print("representative_set_of_primes = ", representative_set_of_primes);
print("conjugate_set_of_primes = ", conjugate_set_of_primes);


print("class_group_cyc = ", class_group_cyc);
